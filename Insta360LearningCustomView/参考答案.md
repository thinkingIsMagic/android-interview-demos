## 一、TimelineView

思路:View 本身只是个"窗口"(宽度 = 屏幕宽,EXACTLY),内容(整条时间轴)比窗口宽,
靠 `scrollX` 横向滚动。所有"时间 ↔ 像素"换算集中到几个小工具方法里,其它逻辑都调它们。

### 1. 先补几个常量 / 字段 / 工具方法

放在现有字段附近即可:

```kotlin
// —— 新增常量 ——
private val basePxPerSec = dp(60f)      // 基准:每秒 60dp 宽
private val rulerHeight = dp(24f)       // 顶部刻度区高度
private val handleHalfWidth = dp(8f)    // trim 手柄半宽
private val minTrimMs = 1000L           // 选区最小时长,防止两个手柄叠在一起

// —— 新增画笔:选区外压暗用 ——
private val dimPaint = Paint().apply { color = 0x99000000.toInt() }

// —— 触摸相关 ——
private val touchSlop = ViewConfiguration.get(context).scaledTouchSlop
private var downX = 0f
private var downY = 0f
private var lastX = 0f
private enum class DragMode { NONE, LEFT_HANDLE, RIGHT_HANDLE, SCROLL }
private var dragMode = DragMode.NONE

// —— 时间/像素换算工具 ——
private fun pxPerMs() = basePxPerSec * scale / 1000f
private fun contentWidthPx() = (clip?.durationMs ?: 0L) * pxPerMs()
private fun timeToX(timeMs: Long) = timeMs * pxPerMs() - scrollX
private fun maxScrollX() = maxOf(0f, contentWidthPx() - width)

private fun formatTime(ms: Long): String {
    val totalSec = ms / 1000
    return "%02d:%02d".format(totalSec / 60, totalSec % 60)
}
```

### 2. onMeasure

```kotlin
override fun onMeasure(widthMeasureSpec: Int, heightMeasureSpec: Int) {
    val widthMode = MeasureSpec.getMode(widthMeasureSpec)
    val widthSize = MeasureSpec.getSize(widthMeasureSpec)
    val heightMode = MeasureSpec.getMode(heightMeasureSpec)
    val heightSize = MeasureSpec.getSize(heightMeasureSpec)

    // 内容理想宽度(整条轴铺开),通常远大于屏幕
    val desiredWidth = contentWidthPx().toInt() + paddingLeft + paddingRight
    val desiredHeight = dp(120f).toInt() + paddingTop + paddingBottom

    val width = when (widthMode) {
        MeasureSpec.EXACTLY -> widthSize                 // match_parent / 固定值:听父的
        MeasureSpec.AT_MOST -> minOf(desiredWidth, widthSize)
        else -> desiredWidth                             // UNSPECIFIED
    }
    val height = when (heightMode) {
        MeasureSpec.EXACTLY -> heightSize
        MeasureSpec.AT_MOST -> minOf(desiredHeight, heightSize)
        else -> desiredHeight
    }
    setMeasuredDimension(width, height)
}
```

> 注:布局里宽是 `match_parent`(EXACTLY),所以 View 不会被撑宽,内容靠 `scrollX` 滚——这正是"窗口"思路。

### 3. onDraw

```kotlin
override fun onDraw(canvas: Canvas) {
    super.onDraw(canvas)
    val clip = this.clip ?: return
    val thumbs = clip.thumbnails
    val trackTop = rulerHeight

    // ① 缩略图轨道:每张色块从自己 timeMs 画到下一张 timeMs(最后一张画到 durationMs)
    val fm = textPaint.fontMetrics
    for (i in thumbs.indices) {
        val startMs = thumbs[i].timeMs
        val endMs = if (i + 1 < thumbs.size) thumbs[i + 1].timeMs else clip.durationMs
        val left = timeToX(startMs)
        val right = timeToX(endMs)
        if (right < 0 || left > width) continue          // 不可见就跳过(性能)

        thumbPaint.color = thumbs[i].color
        canvas.drawRect(left, trackTop, right, height.toFloat(), thumbPaint)

        // 序号写在色块中央
        val cx = (left + right) / 2f
        val cy = (trackTop + height) / 2f
        val baseline = cy - (fm.ascent + fm.descent) / 2f
        canvas.drawText(thumbs[i].index.toString(), cx, baseline, textPaint)
    }

    // ② 刻度尺 + ③ 时间文字:每秒一根竖线,每 5 秒标一次时间
    val durationSec = (clip.durationMs / 1000).toInt()
    for (sec in 0..durationSec) {
        val x = timeToX(sec * 1000L)
        if (x < 0 || x > width) continue
        canvas.drawLine(x, dp(16f), x, rulerHeight, rulerPaint)
        if (sec % 5 == 0) {
            canvas.drawText(formatTime(sec * 1000L), x, dp(12f), textPaint)
        }
    }

    // ④ trim 选区:外部压暗 + 左右手柄
    val leftX = timeToX(clip.trimStartMs)
    val rightX = timeToX(clip.trimEndMs)
    canvas.drawRect(0f, trackTop, leftX, height.toFloat(), dimPaint)            // 左侧压暗
    canvas.drawRect(rightX, trackTop, width.toFloat(), height.toFloat(), dimPaint) // 右侧压暗
    canvas.drawRect(leftX - handleHalfWidth, trackTop, leftX + handleHalfWidth, height.toFloat(), handlePaint)
    canvas.drawRect(rightX - handleHalfWidth, trackTop, rightX + handleHalfWidth, height.toFloat(), handlePaint)
}
```

### 4. onInterceptTouchEvent(外部拦截法)

```kotlin
override fun onInterceptTouchEvent(ev: MotionEvent): Boolean {
    when (ev.actionMasked) {
        MotionEvent.ACTION_DOWN -> {
            downX = ev.x
            downY = ev.y
            return false                          // DOWN 绝不拦截,否则拿不到后续事件
        }
        MotionEvent.ACTION_MOVE -> {
            val dx = abs(ev.x - downX)
            val dy = abs(ev.y - downY)
            if (dx > touchSlop && dx > dy) return true   // 判定为水平滑动 → 拦下来自己处理
        }
    }
    return super.onInterceptTouchEvent(ev)
}
```

### 5. onTouchEvent

```kotlin
override fun onTouchEvent(event: MotionEvent): Boolean {
    val clip = this.clip ?: return super.onTouchEvent(event)
    when (event.actionMasked) {
        MotionEvent.ACTION_DOWN -> {
            lastX = event.x
            val leftX = timeToX(clip.trimStartMs)
            val rightX = timeToX(clip.trimEndMs)
            dragMode = when {
                abs(event.x - leftX) <= handleHalfWidth * 2 -> DragMode.LEFT_HANDLE
                abs(event.x - rightX) <= handleHalfWidth * 2 -> DragMode.RIGHT_HANDLE
                else -> DragMode.SCROLL
            }
            parent?.requestDisallowInterceptTouchEvent(true)   // 占住手势,别让父容器抢
            return true
        }
        MotionEvent.ACTION_MOVE -> {
            val dx = event.x - lastX
            lastX = event.x
            when (dragMode) {
                DragMode.SCROLL -> {
                    scrollX = (scrollX - dx).coerceIn(0f, maxScrollX())
                }
                DragMode.LEFT_HANDLE -> {
                    val dtMs = (dx / pxPerMs()).toLong()
                    val newStart = (clip.trimStartMs + dtMs)
                        .coerceIn(0L, clip.trimEndMs - minTrimMs)
                    this.clip = clip.copy(trimStartMs = newStart)
                }
                DragMode.RIGHT_HANDLE -> {
                    val dtMs = (dx / pxPerMs()).toLong()
                    val newEnd = (clip.trimEndMs + dtMs)
                        .coerceIn(clip.trimStartMs + minTrimMs, clip.durationMs)
                    this.clip = clip.copy(trimEndMs = newEnd)
                }
                DragMode.NONE -> {}
            }
            invalidate()
            return true
        }
        MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
            dragMode = DragMode.NONE
            return true
        }
    }
    return super.onTouchEvent(event)
}
```

> 进阶(没要求,做完上面再玩):
> - **fling 惯性滑动**:`VelocityTracker` 算抬手速度 + `OverScroller.fling()` + `computeScroll()`。
> - **双指缩放 scale**:`ScaleGestureDetector`,在 `onScale` 里改 `scale` 并 `requestLayout()` + `invalidate()`。

import com.bumptech.glide.Glide

private fun loadBasic(target: ImageView) {
    Glide.with(this)

        .into(target)

private fun loadWithPlaceholderAndError(target: ImageView) {
    Glide.with(this)
        .load(GlideSampleData.BROKEN_IMAGE_URL)        // 坏地址 → 最终落到 error 图
        .placeholder(R.drawable.placeholder_gray)       // 加载中显示

        .into(target)

private fun loadCircleCrop(target: ImageView) {
    Glide.with(this)
        .load(GlideSampleData.SINGLE_IMAGE_URL)
        .circleCrop()
        .into(target)
}

private fun loadRounded(target: ImageView) {
    val radiusPx = (24 * resources.displayMetrics.density).toInt()  // 24dp → px
    Glide.with(this)
        .load(GlideSampleData.SINGLE_IMAGE_URL)
        .transform(RoundedCorners(radiusPx))
        .into(target)
}

### GlideImageAdapter.onBindViewHolder

```kotlin
override fun onBindViewHolder(holder: VH, position: Int) {
    val url = urls[position]
    Glide.with(holder.itemView)                         // 绑到 item 的生命周期
        .load(url)
        .placeholder(R.drawable.placeholder_gray)
        .into(holder.binding.ivItem)
}
```

**两个思考题的答案:**

- **为什么用 `holder.itemView`?** `Glide.with()` 的参数决定请求绑哪个生命周期。绑到 itemView(View 版本)
  能跟随其 attach/detach 自动管理请求,比直接传 Activity 更贴合列表项。
- **快速滑动为什么不错图?** ViewHolder 复用时,Glide 在同一个 `ImageView`(target)上发起新请求前,
  会自动 **cancel/clear 掉这个 target 上的旧请求**,所以旧图不会回填到被复用的格子上。

> 进阶:`.diskCacheStrategy(DiskCacheStrategy.ALL)` 控制磁盘缓存;`.override(w, h)` 控制解码尺寸省内存;
> `.thumbnail(0.1f)` 先加载缩略图。
